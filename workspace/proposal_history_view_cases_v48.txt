const serverRow48={proposal_id:'a'.repeat(32),state:'PROPOSAL_READY',proposal_revalidated:false};
await test('48 History discovery has explicit load and metadata-only result labels',()=>{let loaded=0;const tree=render({...candidateState47,proposalItems:[serverRow48],proposalsLoaded:true},{onProposalsRefresh:()=>loaded++});walk(tree).find(n=>n.type==='button'&&label(n)==='Load saved proposals').props.onClick();assert.equal(loaded,1);assert(label(tree).includes('Candidate not rechecked'));});
await test('48 Saved metadata has separate full inspection and recovery actions',()=>{let inspected,recovered;const tree=render({...candidateState47,proposalItems:[serverRow48],proposalsLoaded:true},{onProposalInspect:id=>inspected=id,onProposalRecover:id=>recovered=id});walk(tree).find(n=>n.type==='button'&&label(n)==='Inspect saved proposal').props.onClick();walk(tree).find(n=>n.type==='button'&&label(n)==='Recover saved proposal').props.onClick();assert.equal(inspected,serverRow48.proposal_id);assert.equal(recovered,serverRow48.proposal_id);});
await test('48 Paging is explicit and empty means a loaded empty server page',()=>{let more=0;const tree=render({...candidateState47,proposalItems:[serverRow48],proposalsLoaded:true,proposalNextCursor:serverRow48.proposal_id},{onProposalsMore:()=>more++});walk(tree).find(n=>n.type==='button'&&label(n)==='Load more proposals').props.onClick();assert.equal(more,1);const html=renderToStaticMarkup(render({...candidateState47,proposalItems:[],proposalsLoaded:true}));assert(html.includes('No saved code proposals'));});
await test('48 Proposal history controls freeze during inference or cancellation',()=>{const tree=render({...candidateState47,proposalBusy:true,proposalItems:[serverRow48],proposalsLoaded:true});assert(walk(tree).find(n=>n.type==='button'&&label(n)==='Load saved proposals').props.disabled);assert(walk(tree).find(n=>n.type==='button'&&label(n)==='Inspect saved proposal').props.disabled);});
